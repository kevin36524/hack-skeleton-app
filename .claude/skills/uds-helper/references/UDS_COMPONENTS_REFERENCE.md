## Usage Examples

### Component Import

```typescript
// Server components
import { Box, VStack, HStack, Text, Divider } from '@yahoo/uds';

// Client components (requires 'use client')
import { Button, IconButton, Badge, Avatar } from '@yahoo/uds';

// Icons
import { Star, Inbox, Trash } from '@yahoo/uds-icons';
```

### Basic Layout

```typescript
<Box backgroundColor="primary" spacing="4" borderRadius="md">
  <VStack gap="3" alignItems="flex-start">
    <Text variant="headline1" color="primary">Title</Text>
    <Text variant="body1" color="secondary">Description</Text>
    <Button variant="primary">Click Me</Button>
  </VStack>
</Box>
```

### With Icons

```typescript
import { Icon, IconButton, Button } from '@yahoo/uds';
import { Star, Inbox } from '@yahoo/uds-icons';

// Icon component
<Icon name={Star} size="md" variant="fill" color="warning" />

// Icon button
<IconButton name={Inbox} variant="tertiary" size="sm" aria-label="Inbox" />

// Button with icon
<Button variant="primary" startIcon={Star}>Favorite</Button>
```

### Theme Setup

```typescript
import { Box, DARK_COLOR_MODE_CLASSNAME, LIGHT_COLOR_MODE_CLASSNAME } from '@yahoo/uds';

<Box
  className={`uds-mytheme ${isDarkMode ? DARK_COLOR_MODE_CLASSNAME : LIGHT_COLOR_MODE_CLASSNAME}`}
  backgroundColor="primary"
>
  {/* Your themed content */}
</Box>
```
